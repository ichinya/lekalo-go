	row = map[string]any{"id": id}
	defer func() {
		if r := recover(); r != nil {
			token, ok := r.(string)
			if !ok {
				token = "runtime-error"
			}
			delete(row, "value")
			row["error"] = token
		}
	}()

	// same clock-first precedence as the reference decode.

		}
